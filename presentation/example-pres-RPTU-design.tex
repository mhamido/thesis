\section{Introduction}
\begin{frame}{Introduction: I}
\begin{itemize}
    \item A \textbf{specification} describes the behavior of a programming language:
        \begin{itemize}
            \item Documents in plain text describing it (e.g, the JLS).
            \item A set of evaluation rules (e.g, the semantics of WASM).
        \end{itemize}
    \item An \textbf{implementation} ideally adheres to the specification.
    \item Deviation from the specification (usually) implies the existence of a bug.
    \begin{center}
        \large\textbf{But how would one check that no such mismatch exists?}
    \end{center}
\end{itemize}
\vspace{0.6em}

\end{frame}

\begin{frame}{Introduction: II}

Adherence can be verified through either:
\begin{itemize}
    \item Manual reviews
    \item Testing against a standard test suite, and (optionally) compare results against a \emph{reference implementation}.
\end{itemize}

\vspace{0.8em}
\begin{center}
    \large\textbf{Testing does not prove the \emph{absence} of bugs.}
\end{center}
\end{frame}

\begin{frame}{Thesis}
\begin{itemize}
    \item Encode the specification (semantics) in a dependently-typed language (Agda).
    \item Derive an abstract machine (interpreter) from the semantics.
    \item \emph{Prove} that they are consistent: produce the same results on all programs.\footnote{For programs that terminate.}
\end{itemize}
\end{frame}

\section{Background}
\begin{frame}[fragile]{Background: Agda}
\begin{itemize}
    \item Agda is a dependently typed programming language and proof assistant.
    \item Key distinction: \textbf{parameters} (before the colon) vs \textbf{indices} (after the colon).
\end{itemize}
\vspace{0.5ex}
\begin{minted}{agda}
data Vec (A : Set) : ℕ → Set where
    []  : Vec A zero
    _∷_ : A → Vec A n → Vec A (suc n)
\end{minted}
\begin{itemize}
    \item \textbf{Parameters}: fixed across all constructors.
    \item \textbf{Indices}: each constructor may specialise it to a specific value.
\end{itemize}
\end{frame}

\section{Case Study I: Arithmetic}

%% 1. Syntax
\begin{frame}[fragile]{Arithmetic: Syntax and Intrinsic Typing}
\begin{minted}{agda}
data Type : Set where
    Nat  : Type
    Bool : Type

data ⊢_ : Type → Set where
    `_            : ⊨ T → ⊢ T
    _⊞_           : ⊢ Nat → ⊢ Nat → ⊢ Nat
    test          : ⊢ Nat → ⊢ Bool
    if_then_else_ : ⊢ Bool → ⊢ T → ⊢ T → ⊢ T
\end{minted}
\begin{itemize}
    \item The quote constructor \texttt{`} embeds values (naturals and booleans) into syntax.
    \item Intrinsically typed: only well-typed terms can be constructed.
\end{itemize}
\end{frame}

%% 2. Big-Step
\begin{frame}[fragile]{Arithmetic: Big-Step Semantics}
\begin{minted}{agda}
data _⇓_ : ⊢ T → ⊨ T → Set where
    `_      : (v : ⊨ T) → (` v) ⇓ v
    _⊞_     : e₁ ⇓ n₁ → e₂ ⇓ n₂ → (e₁ ⊞ e₂) ⇓ (n₁ + n₂)
    test    : e ⇓ n → (test e) ⇓ (test' n)
    if-then : e ⇓ true  → e₁ ⇓ v → (if e then e₁ else e₂) ⇓ v
    if-else : e ⇓ false → e₂ ⇓ v → (if e then e₁ else e₂) ⇓ v
\end{minted}
\begin{itemize}
    \item A derivation tree of \texttt{e ⇓ v} is a proof that \texttt{e} evaluates to \texttt{v}.
    \item Properties: \textbf{determinism} and \textbf{totality}.
\end{itemize}
\end{frame}

%% 3. Determinism
\begin{frame}[fragile]{Arithmetic: Determinism}
\begin{minted}{agda}
deterministic : e ⇓ v₁ → e ⇓ v₂ → v₁ ≡ v₂
\end{minted}
\begin{itemize}
    \item Proof by simultaneous induction on both derivations.
    \item Most cases follow immediately: matching constructors imply matching sub-derivations.
    % \item The conditional is the only delicate case: \texttt{if-then} against \texttt{if-else} is a contradiction --- the guard cannot evaluate to both \texttt{true} and \texttt{false}.
\end{itemize}
\end{frame}

%% 4. Totality
\begin{frame}[fragile]{Arithmetic: Totality}
\begin{minted}{agda}
total : ∀ (e : ⊢ T) → ∃ (λ v → e ⇓ v)
\end{minted}
\begin{itemize}
    \item Proof by structural induction on \texttt{e}.
    \item Produce the value and the derivation tree from recursive calls onto the subexpressions
    % \item Example: \texttt{total (e₁ ⊞ e₂)} obtains \texttt{n₁ ﹐ p₁} and \texttt{n₂ ﹐ p₂}, then returns \texttt{n₁ + n₂ ﹐ p₁ ⊞ p₂}.
\end{itemize}
\end{frame}

%% 5. Machine: Frames and Stack
\begin{frame}[fragile]{Arithmetic: Abstract Machine}
\begin{minted}{agda}
data Frame : Type → Type → Set where
    _𝚎𝚕𝚜𝚎_ : ⊢ T → ⊢ T → Frame T    Bool
    _⊞₀_   : Hole → ⊢ Nat → Frame Nat Nat
    _⊞₁_   : ⊨ Nat → Hole → Frame Nat Nat
    test   : Hole → Frame Bool         Nat

data Stack (Answer : Type) : Type → Set where
    []  : Stack Answer Answer
    _∷_ : Stack Answer T₁ → Frame T₁ T₂ → Stack Answer T₂
\end{minted}
\begin{itemize}
    \item \texttt{Frame $T_1$ $T_2$}: indexed by the \textbf{result} type of the term, and the expected type of the \textbf{hole}.
    \item \texttt{Stack Answer T}: a sequence of frames leading from the current focus type \texttt{T} back to the final answer type.
\end{itemize}
\end{frame}

%% 6. Machine: States and Transitions
\begin{frame}[fragile]{Arithmetic: Abstract Machine}
\begin{minted}{agda}
data State (Answer : Type) : Set where
    ⊢_↑_ : Stack Answer T → ⊢ T → State Answer
    ⊢_↓_ : Stack Answer T → ⊨ T → State Answer

data _⇾_ : State Answer → State Answer → Set where
  step-`  : ⊢ stack ↑ ` v             ⇾  ⊢ stack ↓ v
  step-⊞₁ : ⊢ stack ↑ (e₁ ⊞ e₂)       ⇾  ⊢ (stack ∷ ◌ ⊞₀ e₂) ↑ e₁
  step-⊞₂ : ⊢ (stack ∷ ◌ ⊞₀ e₂) ↓ v₁  ⇾  ⊢ (stack ∷ v₁ ⊞₁ ◌) ↑ e₂
  step-⊞₃ : ⊢ (stack ∷ v₁ ⊞₁ ◌) ↓ v₂  ⇾  ⊢ stack ↓ (v₁ + v₂)
  ...
\end{minted}
\begin{itemize}
    \item $\uparrow$: evaluate an expression.
    \item $\downarrow$: return a value.
    % \item The final state \texttt{⊢ [] ↓ v} has an empty stack --- no transition applies.
\end{itemize}
\end{frame}

%% 7. Machine: Determinism
\begin{frame}[fragile]{Arithmetic: Machine Determinism}
\begin{minted}{agda}
det : (s₀ ⇾ s₁) → (s₀ ⇾ s₂) → s₁ ≡ s₂
\end{minted}
\begin{itemize}
    \item At every evaluation step, there is exactly one transition the machine can take.
    \item Proof by case distinction on both transitions.
\end{itemize}
\end{frame}

%% 7b. Machine: Confluence
\begin{frame}[fragile]{Arithmetic: Machine Confluence}
\begin{minted}{agda}
stuck : (⊢ [] ↓ v) ⇾ s → ⊥

confluence : (s ⇾⃰ s₁) → (s ⇾⃰ s₂)
    → (∀ {s'} → s₁ ⇾ s' → ⊥) → (∀ {s'} → s₂ ⇾ s' → ⊥)
    → s₁ ≡ s₂
\end{minted}
\begin{itemize}
    \item A lifted version of \texttt{det}.
    \item Proof by induction on the two traces:
    \begin{itemize}
        \item Both \texttt{base}: trivial.
        \item Both \texttt{step}: apply \texttt{det} on the single step; recursive call on the tails.
        \item One \texttt{step}, one \texttt{base}: the stuck hypothesis contradicts the assumption that $s_1$ and $s_2$ are final states.
    \end{itemize}
    % \item Used in \texttt{correct} to equate the canonical run to $v'$ with the given run to $v$.
\end{itemize}
\end{frame}

%% 8. Completeness
\begin{frame}[fragile]{Arithmetic: Completeness}
\begin{minted}{agda}
complete : e ⇓ v → (⊢ stack ↑ e) ⇾⃰ (⊢ stack ↓ v)
\end{minted}
\begin{itemize}
    \item Evaluating an expression $e$ in a set context, returns $v$. 
    \item Proof by induction on the big-step derivation \texttt{e ⇓ v}.
    \item Each big-step constructor corresponds to a sequence of machine transistions.
\end{itemize}
\end{frame}

%% 9. Completeness: if case
\begin{frame}[fragile]{Arithmetic: Completeness}
\begin{minted}{agda}
complete (if e then e₁ else e₂) v stack (if-then p p₁) =
  (stack ↑ e₀ ⊞ e₁)
  ⇾⟨ step-⊞₀ ⟩                           stack ∷ ◌ ⊞₀ e₁ ↑ e₀
  ⇾⃰⟨ complete (stack ∷ ◌ ⊞₀ e₁) .. ⟩     stack ∷ ◌ ⊞₀ e₁ ↓ _
  ⇾⟨ step-⊞₁ ⟩                           stack ∷ _ ⊞₁ ◌ ↑ e₁
  ⇾⃰⟨ complete (stack ∷ _ ⊞₁ ◌) e₁ _ p₁ ⟩ stack ∷ _ ⊞₁ ◌ ↓ _
  ⇾⟨ step-⊞₂ ⟩                           stack ↓ v
\end{minted}
\end{frame}

%% 10. Correctness
\begin{frame}[fragile]{Arithmetic: Correctness}
\begin{minted}{agda}
correct : (⊢ [] ↑ e) ⇾⃰ (⊢ [] ↓ v) → e ⇓ v
\end{minted}
\begin{itemize}
    \item By totality, obtain some \texttt{v'} with \texttt{e ⇓ v'}.
    \item By completeness, build a canonical run \texttt{⊢ [] ↑ e ⇾⃰ ⊢ [] ↓ v'}.
    \item Confluence of the canonical and given run yields \texttt{v' ≡ v}.
    \item Transport \texttt{e ⇓ v'} along the equality to obtain \texttt{e ⇓ v}.
\end{itemize}
\end{frame}

%% 11. Small-Step
\begin{frame}[fragile]{Arithmetic: Small-Step Semantics}
\begin{minted}{agda}
data _⇾_ : ⊢ T → ⊢ T → Set where
    ξ-⊞₁     : e₁ ⇾ e₁' → (e₁ ⊞ e₂) ⇾ (e₁' ⊞ e₂)
    ξ-⊞₂     : e₂ ⇾ e₂' → (` n₁ ⊞ e₂) ⇾ (` n₁ ⊞ e₂')
    β-⊞      : (` n₁ ⊞ ` n₂) ⇾ ` (n₁ + n₂)
    ξ-if     : e ⇾ e' → (if e then e₁ else e₂) ⇾ (if e' then e₁ else e₂)
    β-if-then : (if ` true  then e₁ else e₂) ⇾ e₁
    β-if-else : (if ` false then e₁ else e₂) ⇾ e₂
    ...
\end{minted}
\begin{itemize}
    \item $\xi$-rules: reduce a sub-expression in context.
    \item $\beta$-rules: perform a single computation step.
    \item Reflexive-transitive closure \texttt{⇾⃰ }.
\end{itemize}
\end{frame}

%% 12. Big-Step <-> Small-Step
\begin{frame}[fragile]{Arithmetic: Big-Step $\leftrightarrow$ Small-Step}
\begin{minted}{agda}
simulate  : e ⇓ v  →  e ⇾⃰  (` v)
simulateᴿ : e ⇾⃰  (` v)  →  e ⇓ v
\end{minted}
\begin{itemize}
    \item \textbf{\texttt{simulate}}: induction on the derivation; apply lifted $\xi$-rules, and then the appropriate $\beta$-rule.
    \item \textbf{\texttt{simulateᴿ}}: via totality + \texttt{simulate} + confluence (values \texttt{` v} are stuck).
\end{itemize}

\begin{minted}{agda}
⇾⃰-⊞₁ : ∀ {e₁ e₁' e₂ : ⊢ Nat} → e₁ ⇾⃰ e₁' → (e₁ ⊞ e₂) ⇾⃰ (e₁' ⊞ e₂)
⇾⃰-⊞₁ base       = base
⇾⃰-⊞₁ (step r p) = step (ξ-⊞₁ r) (⇾⃰-⊞₁ p)

⇾⃰-⊞₂ : ∀ {n : ℕ} {e₂ e₂' : ⊢ Nat} → e₂ ⇾⃰ e₂' → (` n ⊞ e₂) ⇾⃰ (` n ⊞ e₂')
\end{minted}


\end{frame}

%% 13. Denotational
\begin{frame}[fragile]{Arithmetic: Denotational Semantics}
\begin{minted}{agda}
⟦_⟧ : ⊢ T → ⊨ T
⟦ ` v ⟧                  = v
⟦ e₁ ⊞ e₂ ⟧             = ⟦ e₁ ⟧ + ⟦ e₂ ⟧
⟦ if e then e₁ else e₂ ⟧ with ⟦ e ⟧
... | true  = ⟦ e₁ ⟧
... | false = ⟦ e₂ ⟧

simulate  : e ⇓ v  →  ⟦ e ⟧ ≡ v
simulateᴿ : ⟦ e ⟧ ≡ v  →  e ⇓ v
\end{minted}
\begin{itemize}
    \item \texttt{⟦\_⟧} evaluates expressions recursively.
    \item \texttt{simulate}: by induction on the derivation tree.
    \item \texttt{simulateᴿ}: by induction on the expression.
\end{itemize}
\end{frame}

\section{Case Study II: Exceptions}

%% 1. Syntax
\begin{frame}[fragile]{Exceptions: Syntax}
\begin{minted}{agda}
data Exception : Set where
    err : Exception

data ⊢_ : Type → Set where
    ...
    throw         : Exception → ⊢ T
    try_catch_    : ⊢ T → (Exception → ⊢ T) → ⊢ T
\end{minted}
\begin{itemize}
    \item Evaluation can now produce a value \emph{or} raise an exception.
    \item \texttt{throw} raises an exception; \texttt{try-catch} evaluates the body and handles an exception if thrown.
\end{itemize}
\end{frame}

%% 2. Big-Step Semantics
\begin{frame}[fragile]{Exceptions: Big-Step Semantics}
\begin{minted}{agda}
data Result (T : Type) : Set where
    return : ⊨ T → Result T
    raise  : Exception → Result T

e ⇓ v  = e ⇓ʳ return v
e ⇑ ex = e ⇓ʳ raise ex

data _⇓ʳ_ : ⊢ T → Result T → Set where
    _⊞_        : e₁ ⇓ n₁ → e₂ ⇓ n₂ → (e₁ ⊞ e₂) ⇓ (n₁ + n₂)
    ⊞-exn₁     : e₁ ⇑ ex           → (e₁ ⊞ e₂) ⇑ ex
    ⊞-exn₂     : e₁ ⇓ n₁ → e₂ ⇑ ex → (e₁ ⊞ e₂) ⇑ ex
    ...
    throw-rule : throw ex ⇑ ex
    try-return : e ⇓ v → (try e catch h) ⇓ v
    try-catch  : e ⇑ ex → h ex ⇓ʳ r → (try e catch h) ⇓ʳ r 
\end{minted}
\begin{itemize}
    \item Each constructor gains additional exception-propagation rules (e.g.\ \texttt{⊞-exn₁}).
\end{itemize}
\end{frame}

%% 3. Totality
\begin{frame}[fragile]{Exceptions: Totality}
\begin{minted}{agda}
total : ∀ (e : ⊢ T) → ∃ (λ r → e ⇓ʳ r)
\end{minted}
\begin{itemize}
    \item Follows the same structure as arithmetic: structural induction on \texttt{e}.
    \item The result is now a \texttt{Result T}: either \texttt{return v} or \texttt{raise ex}.
\end{itemize}
\vspace{0.25em}
\begin{minted}{agda}
total (throw ex) = raise ex ﹐ throw-rule

total (try e catch h) with total e
... | return v  ﹐ p = return v ﹐ try-return p
... | raise ex  ﹐ p = let (r ﹐ q) = total (h ex) in r ﹐ try-catch p q
\end{minted}
\end{frame}

%% 4. Machine: Frames
\begin{frame}[fragile]{Exceptions: Abstract Machine}
\begin{minted}{agda}
data Frame : Type → Type → Set where
    ...
    catch   : (Exception → ⊢ T) → Frame T T
\end{minted}
\begin{itemize}
    \item \texttt{catch h}: stores the exception handler.
    \item If the body raises an exception, frames should be discarded until the nearest handler is found.
\end{itemize}
\end{frame}

%% 5. Machine: States and Transitions
\begin{frame}[fragile]{Exceptions: Abstract Machine}
\begin{minted}{agda}
data State (Answer : Type) : Set where
    ⊢_↑_ : Stack Answer T → ⊢ T      → State Answer
    ⊢_↓_ : Stack Answer T → ⊨ T      → State Answer
    ⊢_☇_ : Stack Answer T → Exception → State Answer

step-throw     : ⊢ stack ↑ throw ex        ⇾  ⊢ stack ☇ ex
step-try       : ⊢ stk ↑ (try e catch h)   ⇾  ⊢ stk ∷ catch h ↑ e
step-catch-ok  : ⊢ stk ∷ catch h ↓ v       ⇾  ⊢ stk ↓ v
step-catch-exn : ⊢ stack ∷ catch h ☇ ex    ⇾  ⊢ stack ↑ h ex
step-drop-⊞₀   : ⊢ stack ∷ (◌ ⊞₀ e₂) ☇ ex  ⇾  ⊢ stack ☇ ex
\end{minted}
\begin{itemize}
    \item Unwind mode propagates the exception, and scans the stack for the nearest catch frame
    \item Final states: \texttt{⊢ [] ↓ v} (return) and \texttt{⊢ [] ☇ ex} (unhandled exception).
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Exceptions: Abstract Machine}
\begin{itemize}
    \item Why not use a generic \texttt{step-drop} rule that discards any frame?
\end{itemize}
\begin{minted}{agda}
    step-discard    : ∀ {f : Frame T₁ T₂}
                      ⊢ stack ∷ f ☇ ex        ⇾ ⊢ stack ☇ ex

    step-catch-exn  : ⊢ stack ∷ catch h ☇ ex  ⇾ ⊢ stack ↑ h ex
\end{minted}

\begin{itemize}
    \item Causes the abstract machine to be non-deterministic. 
\end{itemize}

\end{frame}


%% 6. Machine: Determinism
\begin{frame}[fragile]{Exceptions: Machine Determinism}
\begin{minted}{agda}
det : (s₀ ⇾ s₁) → (s₀ ⇾ s₂) → s₁ ≡ s₂
\end{minted}
\begin{itemize}
    \item At every evaluation step, there is exactly one transition the machine can take.
    \item Proof by case distinction on both transitions.
\end{itemize}
\end{frame}

%% 6b. Machine: Confluence
\begin{frame}[fragile]{Exceptions: Machine Confluence}
\begin{minted}{agda}
stuck-↓ : (⊢ [] ↓ v)  ⇾ s → ⊥
stuck-☇ : (⊢ [] ☇ ex) ⇾ s → ⊥
↓≢☇     : (⊢ [] ↓ v) ≡ (⊢ [] ☇ ex) → ⊥

confluence : (s ⇾⃰ s₁) → (s ⇾⃰ s₂)
    → (∀ {s'} → s₁ ⇾ s' → ⊥) → (∀ {s'} → s₂ ⇾ s' → ⊥)
    → s₁ ≡ s₂
\end{minted}
\begin{itemize}
    \item A lifted version of \texttt{det}: same three-case induction as Arithmetic.
    \item There are now \emph{two} final states.
\end{itemize}
\end{frame}

%% 7. Completeness
\begin{frame}[fragile]{Exceptions: Completeness}
\begin{minted}{agda}
complete     : e ⇓ v  → (⊢ stk ↑ e) ⇾⃰ (⊢ stk ↓ v)
complete-exn : e ⇑ ex → (⊢ stk ↑ e) ⇾⃰ (⊢ stk ☇ ex)
\end{minted}
\begin{itemize}
    \item Two mutually recursive lemmas: one for each final state.
    \item \texttt{try-return}: body succeeds; \texttt{complete} evaluates it, \texttt{step-catch-ok} discards the handler frame.
    \item \texttt{try-catch}: \texttt{complete-exn} drives the body to \verb|☇|; \texttt{step-catch-exn} activates the handler; \texttt{complete} finishes.
\end{itemize}
\end{frame}

%% 8. Completeness: try-catch case
\begin{frame}[fragile]{Exceptions: Completeness}
\begin{minted}{agda}
complete (try e catch h) v stk (try-catch {ex = ex} pe ph) =
  step step-try
  (⇾⃰-transitive (complete-exn e ex (stk ∷ catch h) pe)
  (step step-catch-exn
  (complete (h ex) v stk ph)))
\end{minted}
\begin{itemize}
    \item Push \texttt{catch h} frame, transistion to $\uparrow$ mode.
    \item \texttt{complete-exn} drives the body to \verb|☇| at the handler frame.
    \item \texttt{step-catch-exn} activates the handler; \texttt{complete} evaluates it to $\downarrow$.
\end{itemize}
\end{frame}

%% 9. Correctness
\begin{frame}[fragile]{Exceptions: Correctness}
\begin{minted}{agda}
correct     : (⊢ [] ↑ e) ⇾⃰ (⊢ [] ↓ v)  → e ⇓ v
correct-exn : (⊢ [] ↑ e) ⇾⃰ (⊢ [] ☇ ex) → e ⇑ ex

correct e v tr with total e
... | return v' ﹐ p = .. -- complete → confluence → v' ≡ v
... | raise ex  ﹐ p = .. -- complete-exn → confluence → ↓≢☇ → ex-falso
\end{minted}
\begin{itemize}
    \item Totality splits into two cases.
    \item \begin{itemize}
        \item Both are \texttt{↓}-runs; confluence yields \texttt{v' ≡ v}.
        \item Both are \texttt{☇}-runs; confluence yields \texttt{ex' ≡ ex}.
        \item One is an \texttt{☇}-run, the other is a \texttt{↓}-run; confluence yields \texttt{↓ ≡ ☇}, contradiction.
    \end{itemize}
\end{itemize}
\end{frame}

%% 10. Small-Step
\begin{frame}[fragile]{Exceptions: Small-Step Semantics}
\begin{minted}{agda}
data _⇾_ : ⊢ T → ⊢ T → Set where
    ...
    β-⊞-exn₁   : (throw ex ⊞ e₂) ⇾ throw ex
    β-⊞-exn₂   : (` n₁ ⊞ throw ex) ⇾ throw ex
    
    ξ-try       : e ⇾ e' → (try e catch h) ⇾ (try e' catch h)
    β-try-ok    : (try (` v) catch h) ⇾ ` v
    β-try-throw : (try (throw ex) catch h) ⇾ h ex
\end{minted}
\begin{itemize}
    \item $\beta$-exn rules propagate \texttt{throw} upward through each constructor.
    \item \texttt{β-try-throw} is the only rule that \emph{consumes} a \texttt{throw}, activating the handler.
\end{itemize}
\end{frame}

%% 11. Big-Step <-> Small-Step
\begin{frame}[fragile]{Exceptions: Big-Step $\leftrightarrow$ Small-Step}
\begin{minted}{agda}
resultExpr : Result T → ⊢ T
resultExpr (return v) = ` v
resultExpr (raise ex) = throw ex

simulate  : e ⇓ʳ r  →  e ⇾⃰ resultExpr r
simulateᴿ : e ⇾⃰  (resultExpr r)  →  e ⇓ʳ r
\end{minted}
\begin{itemize}
    \item \texttt{resultExpr} maps a result to its stuck expression (\texttt{` v} or \texttt{throw ex}).
    \item \textbf{\texttt{simulate}}: induction on the derivation; exception cases use $\beta$-exn rules to bubble \texttt{throw} up.
    \item \textbf{\texttt{simulateᴿ}}: totality + \texttt{simulate} + confluence + \texttt{resultExpr-inj} then transport.
\end{itemize}
\end{frame}

%% 12. Denotational
\begin{frame}[fragile]{Exceptions: Denotational Semantics}
\begin{minted}{agda}
interpret : ⊢ T → Result T
interpret (throw ex)      = raise ex
interpret (try e catch h) with interpret e
... | raise ex = interpret (h ex)
... | return v = return v
...

simulate  : e ⇓ʳ r  →  interpret e ≡ r
simulateᴿ : interpret e ≡ r  →  e ⇓ʳ r
\end{minted}
\begin{itemize}
    \item \texttt{interpret} propagates exceptions structurally, mirroring big-step.
    \item \texttt{simulate}: by induction; \texttt{rewrite} collapses the equality at each constructor.
    \item \texttt{simulateᴿ}: reads the result from the equality and reconstructs the derivation.
\end{itemize}
\end{frame}

%% 1. Syntax
\begin{frame}[fragile]{STLC: Contexts and Environments}
\begin{itemize}
    \item Contexts record the types of bound variables.
    \item Environments record the values of bound variables of a context.
\end{itemize}
\begin{minted}{agda}
data Context : Set where

    _,_  : Context → Type → Context

data Environment : Context → Set where
    ∅     : Environment ∅
    _,_   : Environment Γ → ⊨ T → Environment (Γ , T)
\end{minted}
\end{frame}

\begin{frame}[fragile]{STLC: De Bruijn Indices}
\begin{minted}{agda}

data _∈_ : Type → Context → Set where

  S_  : T₁ ∈ Γ → T₁ ∈ (Γ , T₂)

lookupₑ : Environment Γ → T ∈ Γ → Value T
lookupₑ (γ , x) Z = x
lookupₑ (γ , x) (S y) = lookupₑ γ y
\end{minted}
\end{frame}


\begin{frame}[fragile]{STLC: Syntax and Intrinsic Typing}
\begin{minted}{agda}
data Type : Set where
    Bool : Type;  Nat : Type;  _⇒_ : Type → Type → Type

data _⊢_ (Γ : Context) : Type → Set where
    Nat  : ℕ → Γ ⊢ Nat;  Bool : 𝔹 → Γ ⊢ Bool
    Var  : T ∈ Γ → Γ ⊢ T
    ƛ_   : (Γ , T₁) ⊢ T₂ → Γ ⊢ (T₁ ⇒ T₂)

% \begin{itemize}

% \end{itemize}
\end{frame}

%% 2. Big-Step Semantics
\begin{frame}[fragile]{STLC: Big-Step Semantics}
\begin{minted}{agda}
record Closure (A B : Type) : Set where
    field {Γᶜ} : Context;  δᶜ : Environment Γᶜ;  eᶜ : (Γᶜ , A) ⊢ B

⊨ (T₁ ⇒ T₂) = Closure T₁ T₂

data _⊢_⇓_ : Environment Γ → Γ ⊢ T → ⊨ T → Set where
    VAR : δ ⊢ Var x ⇓ lookupₑ δ x
    ABS : δ ⊢ ƛ e  ⇓ ⟨ δ , e ⟩
    app : δ ⊢ e₀ ⇓ ⟨ δᶜ , eᶜ ⟩ → δ ⊢ e₁ ⇓ v₁ → (δᶜ , v₁) ⊢ eᶜ ⇓ v
        → δ ⊢ e₀ · e₁ ⇓ v
\end{minted}
\begin{itemize}
    \item Evaluation is \emph{environment-based}: variables resolved by \texttt{lookupₑ δ}.
    \item A $\lambda$ captures the current environment $\delta$ as a closure.
    \item Application evaluates function, argument, then runs the closure body in \texttt{δᶜ , v₁}.
\end{itemize}
\end{frame}

%% 3. Determinism
\begin{frame}[fragile]{STLC: Determinism}
\begin{minted}{agda}
deterministic : δ ⊢ e ⇓ v₁ → δ ⊢ e ⇓ v₂ → v₁ ≡ v₂
\end{minted}
\begin{itemize}
    \item Proof by simultaneous induction on both derivations.
    \item \texttt{NAT}, \texttt{BOOL}, \texttt{VAR}, \texttt{ABS}: trivial.
    \item \texttt{app}: induction applied three times, on function value, argument value, and closure body.
\end{itemize}
\end{frame}

%% 4. Totality
\begin{frame}[fragile]{STLC: Totality}
\begin{minted}{agda}
{-# TERMINATING #-}
total : ∀ (e : Γ ⊢ T) → ∃ (λ v → δ ⊢ e ⇓ v)
total (e₁ · e₂) with total e₁ | total e₂
... | ⟨ δᶜ , eᶜ ⟩ ʻ p | v₂ ʻ q with total {δ = δᶜ , v₂} eᶜ
... | v ʻ r = v ʻ app p q r
\end{minted}
\begin{itemize}
    \item STLC is strongly normalising (always terminates with a value), but Agda cannot verify this structurally.
    % \item The recursive call on \texttt{eᶜ} goes through a closure obtained by evaluating \texttt{e₁} --- not a sub-expression of \texttt{e₁ · e₂}.
    \item The \texttt{\{-\# TERMINATING \#-\}} pragma can be used to disable the check, leaving the burden of proof to the programmer.
\end{itemize}
\end{frame}

%% 5. Machine: Frames
\begin{frame}[fragile]{STLC: Abstract Machine}
\begin{minted}{agda}
data Frame : Type → Type → Set where
    ...
    app₁   : Hole → Γ ⊢ A            → Frame B (A ⇒ B)
    app₂   : ⊨ (A ⇒ B) → Hole        → Frame B A      
    app₃   : Environment Γ → ⊨ (A ⇒ B) → ⊨ A → Hole → Frame B B
\end{minted}
\begin{itemize}
    \item Application decomposes into \emph{three} sequential frames.
    \item \texttt{app₃} stores the \emph{caller's environment} $\delta$ so it can be restored after the call.
    \item mirrors the behaviour in the big-step \texttt{app} rule.
\end{itemize}
\end{frame}

%% 6. Machine: States and Transitions
\begin{frame}[fragile]{STLC: Abstract Machine}
\begin{minted}{agda}
data State (Answer : Type) : Set where
    _⊢_↑_ : Environment Γ → Stack Answer A → Γ ⊢ A → State Answer
    _⊢_↓_ : Environment Γ → Stack Answer A → ⊨ A   → State Answer

step-ƛ  : δ ⊢ stack ↑ ƛ e   ⇾  δ ⊢ stack ↓ ⟨ δ , e ⟩

step-·₂ : δ ⊢ stack ∷ app₂ ⟨ δᶜ , eᶜ ⟩ ◌ ↓ v₂
        ⇾  (δᶜ , v₂) ⊢ stack ∷ app₃ δ ⟨ δᶜ , eᶜ ⟩ v₂ ◌ ↑ eᶜ

step-·₃ : δ ⊢ stack ∷ app₃ δ' clo v₂ ◌ ↓ v₃  ⇾  δ' ⊢ stack ↓ v₃
\end{minted}
\begin{itemize}
    \item Each state carries the \emph{current environment} $\delta$ alongside the stack.
    \item \texttt{step-·₂}: switches to closure environment \texttt{δᶜ , v₂} to evaluate the body.
    \item \texttt{step-·₃}: returns to the caller's environment \texttt{δ'} from the \texttt{app₃} frame.
\end{itemize}
\end{frame}

%% 8. Completeness
\begin{frame}[fragile]{STLC: Completeness}
\begin{minted}{agda}
complete : δ ⊢ e ⇓ v → (δ ⊢ stack ↑ e) ⇾⃰ (δ ⊢ stack ↓ v)
\end{minted}
\begin{itemize}
    \item Proof by induction on the big-step derivation; generalised over an arbitrary \texttt{stack}.
    \item \texttt{app} case: three recursive calls to handle \texttt{step-·}, \texttt{step-·₁}, \texttt{step-·₂}, \texttt{step-·₃}.
    % \item \texttt{step-·₂} switches the environment from $\delta$ to $\delta^c,v_1$ to enter the closure body.
    % \item \texttt{step-·₃} restores $\delta$ from the \texttt{app₃} frame, closing the application trace.
\end{itemize}
\end{frame}

%% 9. Completeness: app case
\begin{frame}[fragile]{STLC: Completeness}
\begin{minted}{agda}
complete δ stack (e₀ · e₁) v (app {δᶜ = δᶜ} {eᶜ} {v₁} p p₁ p₂) =
  proof         δ ⊢ stack ↑ (e₀ · e₁)
  ⇾⟨ step-· ⟩                  δ ⊢ stack ∷ app₁ ◌ e₁ ↑ e₀
  ⇾⃰⟨ complete δ (stack ∷ app₁ ◌ e₁) e₀ ⟨ δᶜ , eᶜ ⟩ p ⟩ ...
  ⇾⟨ step-·₁ ⟩ ... ⇾⃰⟨ complete δ (stack ∷ app₂ _ ◌) e₁ v₁ p₁ ⟩ ...
  ⇾⟨ step-·₂ ⟩
    (δᶜ , v₁) ⊢ stack ∷ app₃ δ ⟨ δᶜ , eᶜ ⟩ v₁ ◌ ↑ eᶜ
  ⇾⃰⟨ complete (δᶜ , v₁) (stack ∷ app₃ δ _ v₁ ◌) eᶜ v p₂ ⟩ ...
  ⇾⟨ step-·₃ ⟩  δ ⊢ stack ↓ v  ∎
\end{minted}
\begin{itemize}
    \item Six transitions; three IHs handle the sub-evaluations.
    \item Environment switches at \texttt{step-·₂} (enter closure) and back at \texttt{step-·₃} (exit).
\end{itemize}
\end{frame}

%% 10. Correctness
\begin{frame}[fragile]{STLC: Correctness}
\begin{minted}{agda}
correct : (δ ⊢ [] ↑ e) ⇾⃰ (δ ⊢ [] ↓ v) → δ ⊢ e ⇓ v
\end{minted}
\begin{itemize}
    \item Follows the same confluence argument.
    \item Obtain \texttt{v', p : $e \Downarrow v$} from \texttt{total e}; build run via \texttt{complete}.
    \item Confluence of both runs run yields \texttt{δ ⊢ [] ↓ v' ≡ δ ⊢ [] ↓ v}.
    \item Injectively, \texttt{v' ≡ v}; therefore we can obtain $e \Downarrow v$.
\end{itemize}
\end{frame}

%% 11. Small-Step
\begin{frame}[fragile]{STLC: Small-Step Semantics}
\begin{minted}{agda}
data Normal : Γ ⊢ T → Set where
    nat : Normal (Nat n);  bool : Normal (Bool b);  closure : Normal (ƛ e)

data _↦_ : Γ ⊢ T → Γ ⊢ T → Set where
    β-⊞         : Nat n₁ ⊞ Nat n₂ ↦ Nat (n₁ + n₂)
    β-app       : ƛ e · v ↦ e [ v ]        
    ξ-app-left  : e₁ ↦ e₁' → e₁ · e₂ ↦ e₁' · e₂
    ξ-app-right : Normal e₁ → e₂ ↦ e₂' → e₁ · e₂ ↦ e₁ · e₂'
    ...
\end{minted}
\begin{itemize}
    \item \textbf{Small-step is Substitution-based}: \texttt{β-app} uses \texttt{\_[\_]}, substituting the argument into the body (shifting indices in the process).
    \item \textbf{Big-step is environment-based}: the rule \texttt{app} extends $\delta$ with the argument $v$.
\end{itemize}
\end{frame}

%% 12. Denotational
\begin{frame}[fragile]{STLC: Denotational Semantics}
\begin{minted}{agda}
evaluate : Environment Γ → Γ ⊢ T → ⊨ T
evaluate δ (ƛ e)      = ⟨ δ , e ⟩
evaluate δ (e₁ · e₂) with evaluate δ e₁
... | ⟨ δᶜ , eᶜ ⟩    = apply ⟨ δᶜ , eᶜ ⟩ (evaluate δ e₂)

apply : ⊨ (A ⇒ B) → ⊨ A → ⊨ B
apply ⟨ δᶜ , eᶜ ⟩ v = evaluate (δᶜ , v) eᶜ

{-# TERMINATING #-}
simulate  : evaluate δ e ≡ v  →  δ ⊢ e ⇓ v
simulateᴿ : δ ⊢ e ⇓ v  →  evaluate δ e ≡ v
\end{minted}
\begin{itemize}
    \item Same \texttt{TERMINATING} caveat: the closure body call is not structurally smaller.
    \item \texttt{simulate}/\texttt{simulateᴿ}: each direction by induction; \texttt{rewrite} at each constructor.
\end{itemize}
\end{frame}

%% 13. Big-Step <-> Small-Step (open)
\begin{frame}[fragile]{STLC: Big-Step $\leftrightarrow$ Small-Step}
\begin{minted}{agda}
  e [ v₁ ] ↦* embed v₂ 
→ δ ⊢ (ƛ e) · v₁ ⇓ v₂
\end{minted}
\begin{itemize}
    \item The \texttt{simulateᴿ} direction for \texttt{app} must convert a small-step trace into a big-step derivation --- but big-step is environment-based, so the result value $v_2$ must already be known.
    \item Alternatively, if can quote values back into syntax:
\end{itemize}

\begin{minted}{agda}
   δ ⊢ s [ ` v ]                  ⇓ v₂  
→  δ ⊢ s [ ` ⟨ δᶜ , eᶜ ⟩ · ` v₁ ] ⇓ v₂
\end{minted}

\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion: Summary}
\begin{itemize}
    \item Presented a systematic process for deriving an abstract machine from big-step semantics for Arithmetic, Exceptions, and STLC.
    \item The derivation proceeds by:
    \begin{itemize}
        \item representing the process of evaluation explicitly via \textbf{frames}.
        \item defining machine states as a \textbf{stack of frames} in three modes: \emph{eval}, \emph{return}, and (for exceptions) \emph{unwind};
        \item defining \textbf{transition rules} by mimicking the in-order evaluation of terms.
    \end{itemize}
    \item \textbf{Completeness}: the machine trace produces the same value as a big-step derivation.
    \item \textbf{Correctness}: totality yields a run of the machine to $v'$; any run to $v$ must converge to the same result.
    \item \textbf{Compositionality}: derive an equivalence between other forms of semantics using big-step as an intermediary.
\end{itemize}
\end{frame}

\begin{frame}{Conclusion: Limitations and Future Work}
\begin{itemize}
    \item Current route assumes totality and does not account for non-terminating programs.
    \item STLC totality currently uses a termination pragma.
    \item STLC small-step $\leftrightarrow$ big-step equivalence is open.
\end{itemize}
\end{frame}
